# 1. 创建 Docker 网络
sudo docker network create iotdb --subnet 172.18.0.0/16

# 2. 启动 IoTDB 容器
cd /home/jetson/iotdb
sudo docker compose -f docker-compose-standalone.yml up -d

# 3. 等待约 60 秒后验证
sudo docker exec iotdb-service /iotdb/sbin/start-cli.sh -h iotdb-service -e 'SHOW VERSION'
